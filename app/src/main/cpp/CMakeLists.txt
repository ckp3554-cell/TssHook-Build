cmake_minimum_required(VERSION 3.22)
project(tsshook)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SHADOWHOOK_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(SHADOWHOOK_BUILD_STATIC ON  CACHE BOOL "" FORCE)
set(SHADOWHOOK_BUILD_TEST   OFF CACHE BOOL "" FORCE)
set(SHADOWHOOK_BUILD_SAMPLE OFF CACHE BOOL "" FORCE)
add_subdirectory(shadowhook shadowhook_build)
add_library(tsshook SHARED tss_hook.cpp)
target_link_libraries(tsshook shadowhook log dl android)
